```python
"""同 IP 24h 时间窗内,跨 ssh + http + smtp 多阶段攻击链识别。"""
from __future__ import annotations
from dataclasses import dataclass
from datetime import datetime, timedelta
from typing import Sequence

from shared_llm_core import (
    Finding, FindingSource, FindingSeverity,
    Rule, RuleContext, RuleEngine, RuleRegistry,
)


class MultiStageCorrelationRule(Rule):
    id = "001-correlate-multistage"
    severity_hint = "high"

    # 默认时间窗 24h
    def __init__(self, window: timedelta = timedelta(hours=24)) -> None:
        self._window = window

    def evaluate(self, ctx: RuleContext) -> list[Finding]:
        events = ctx.facts.get("events", [])
        if not events:
            return []
        # 简化:同 subject(IP),时间窗内 ≥ 3 个不同 protocol → 触发
        protocols = {e.get("protocol") for e in events if e.get("protocol")}
        if len(protocols) < 3:
            return []
        return [Finding(
            id="",
            source=FindingSource.SOC,
            severity=FindingSeverity.HIGH,
            confidence=0.8,
            title=f"Multi-stage attack on {ctx.subject}",
            host=ctx.subject,
            evidence=(f"protocols={sorted(protocols)}", f"events={len(events)}"),
            tags=frozenset(["multi-stage", "correlation"]),
        )]
```

```python
def _select_tier_for_event(log_line: str) -> TaskTier:
    """根据事件 criticality 自动选 LLM tier。
    
    - 含 'critical' / 'CRITICAL' → PREMIUM
    - 含 'failed password' (auth failure) → STANDARD  
    - 普通 INFO 日志 → CHEAP
    """
    upper = log_line.upper()
    if "CRITICAL" in upper:
        return TaskTier.PREMIUM
    if "FAILED PASSWORD" in upper or "AUTH FAILURE" in upper:
        return TaskTier.STANDARD
    return TaskTier.CHEAP
```

```python
"""用 §7 MultiAgentOrchestrator 写 case summary。

角色:
- ANALYST: 分析事件背景
- EXPLOITER: 攻击视角
- SYNTHESIZER: 汇总成 3 段式 narrative
"""
from __future__ import annotations

from shared_llm_core import (
    AgentRole, AgentResult, MissionContext, MultiAgentOrchestrator,
)


def build_case_summary(
    orch: MultiAgentOrchestrator,
    events: list[dict],
    subject: str,
) -> AgentResult:
    mission = MissionContext(
        task=f"Summarize the following security events on {subject}",
        inputs={"subject": subject, "events": events},
    )
    results = orch.run(mission, [AgentRole.ANALYST, AgentRole.EXPLOITER, AgentRole.SYNTHESIZER])
    # 最后一个是 SYNTHESIZER
    return results[-1]
```
