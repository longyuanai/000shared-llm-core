### ISSUE 2 · API 封装层 + 类型

**`src/lib/types.ts`**:

```typescript
// 从 shared-llm-core Finding / Correlation 复制(避免 deep import)
export type FindingSeverity = "info" | "low" | "medium" | "high" | "critical";
export type FindingSource = "soc" | "vuln" | "lab" | "code" | "reverse" | "firmware";

export interface Finding {
  id: string;
  source: FindingSource;
  severity: FindingSeverity;
  confidence: number;
  title: string;
  host?: string;
  ts?: string;            // ISO 8601
  cve?: string;
  narrative?: string;
}

export interface Correlation {
  id: string;
  rule_id: string;
  severity: FindingSeverity;
  findings: string[];      // Finding.id 列表
  narrative: string;
  ts: string;
}

export interface HealthResponse {
  status: "ok" | "degraded";
  products: Record<FindingSource, { status: string; version: string }>;
}
```

**`src/lib/api.ts`**:

```typescript
import type { Finding, Correlation, HealthResponse } from "./types";

const BASE = process.env.NEXT_PUBLIC_GATEWAY_URL ?? "http://localhost:8080";

export async function getHealth(): Promise<HealthResponse> {
  return fetch(`${BASE}/v0.5/health`).then(r => r.json());
}

export async function listFindings(params: {
  source?: FindingSource;
  severity?: FindingSeverity;
  host?: string;
  limit?: number;
}): Promise<Finding[]> {
  const q = new URLSearchParams();
  Object.entries(params).forEach(([k, v]) => v != null && q.set(k, String(v)));
  return fetch(`${BASE}/v0.5/findings?${q}`).then(r => r.json());
}

export async function listCorrelations(): Promise<Correlation[]> {
  return fetch(`${BASE}/v0.5/correlations`).then(r => r.json());
}

export async function triggerScan(
  source: FindingSource,
  payload: Record<string, unknown>,
): Promise<{ scan_id: string }> {
  return fetch(`${BASE}/v0.5/${source}/scan`, {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify(payload),
  }).then(r => r.json());
}
```

### ISSUE 3 · Dashboard 总览页 + 6 个产品 tab

**`src/app/page.tsx`(Dashboard home)**:

```typescript
import { getHealth, listFindings, listCorrelations } from "@/lib/api";
import { Card } from "@/components/ui/card";

export default async function HomePage() {
  const [health, findings, correlations] = await Promise.all([
    getHealth(),
    listFindings({ limit: 100 }),
    listCorrelations(),
  ]);

  const bySource = findings.reduce<Record<string, number>>((acc, f) => {
    acc[f.source] = (acc[f.source] ?? 0) + 1;
    return acc;
  }, {});

  return (
    <main className="p-8">
      <h1 className="text-3xl font-bold">longyuanai Dashboard</h1>
      <p className="text-muted-foreground">
        {Object.values(health.products).every(p => p.status === "ok")
          ? "All products healthy"
          : "Some products degraded"}
      </p>

      <div className="mt-8 grid grid-cols-3 gap-4">
        {(["soc","vuln","lab","code","reverse","firmware"] as const).map(src => (
          <Card key={src} className="p-4">
            <h2 className="text-xl font-semibold uppercase">{src}</h2>
            <p className="text-2xl mt-2">{bySource[src] ?? 0} findings</p>
            <a href={`/${src}`} className="text-blue-400">View →</a>
          </Card>
        ))}
      </div>

      <h2 className="mt-12 text-2xl font-bold">Correlations ({correlations.length})</h2>
      <ul>
        {correlations.slice(0, 10).map(c => (
          <li key={c.id} className="border-b py-2">
            <span className={`badge badge-${c.severity}`}>{c.severity}</span>
            {c.narrative}
          </li>
        ))}
      </ul>
    </main>
  );
}
```

**6 个产品 tab 共用模板**(`src/app/[source]/page.tsx`):

```typescript
import { listFindings } from "@/lib/api";
import { FindingCard } from "@/components/FindingCard";
import { notFound } from "next/navigation";

const VALID = ["001","002","003","004","005","006"] as const;
type Props = { params: { source: string } };

export default async function ProductPage({ params }: Props) {
  if (!VALID.includes(params.source as any)) notFound();
  const sourceMap = { "001":"soc","002":"vuln","003":"lab","004":"code","005":"reverse","006":"firmware" } as const;
  const source = sourceMap[params.source as keyof typeof sourceMap];
  const findings = await listFindings({ source, limit: 50 });

  return (
    <main className="p-8">
      <h1 className="text-3xl font-bold">Product {params.source}</h1>
      <p className="text-muted-foreground">{findings.length} findings</p>
      <div className="mt-4 space-y-2">
        {findings.map(f => <FindingCard key={f.id} finding={f} />)}
      </div>
    </main>
  );
}
```

### ISSUE 4 · SSE 实时流

**`src/app/api/stream/route.ts`(Next.js Route Handler 代理 SSE)**:

```typescript
export const dynamic = "force-dynamic";
export const runtime = "nodejs";

const GATEWAY = process.env.NEXT_PUBLIC_GATEWAY_URL ?? "http://localhost:8080";

export async function GET() {
  const upstream = await fetch(`${GATEWAY}/v0.5/stream`);
  return new Response(upstream.body, {
    headers: {
      "Content-Type": "text/event-stream",
      "Cache-Control": "no-cache",
      "Connection": "keep-alive",
    },
  });
}
```

**`src/components/StreamBadge.tsx`**(客户端组件):

```typescript
"use client";
import { useEffect, useState } from "react";
import { Badge } from "@/components/ui/badge";

export function StreamBadge() {
  const [count, setCount] = useState(0);
  useEffect(() => {
    const es = new EventSource("/api/stream");
    es.addEventListener("finding", () => setCount(c => c + 1));
    es.onerror = () => es.close();
    return () => es.close();
  }, []);
  return <Badge variant="outline">Live: {count} new findings</Badge>;
}
```
