```python
    AgentRole, AgentResult, MissionContext, MultiAgentOrchestrator,
```

```python
"""跨场景关联报告:同一目标被多个场景攻击 → 自动关联。"""
from __future__ import annotations
from shared_llm_core import (
    Finding, FindingSource, FindingSeverity,
    FindingRegistry,
)


def correlate_findings(
    findings: list[Finding],
    *,
    same_host_window_days: int = 1,
) -> list[dict]:
    """简化:同 host 的 Finding 数量 ≥ 2 → 视为关联。"""
    by_host: dict[str, list[Finding]] = {}
    for f in findings:
        if f.host:
            by_host.setdefault(f.host, []).append(f)
    correlations = []
    for host, group in by_host.items():
        if len(group) >= 2:
            correlations.append({
                "host": host,
                "findings": [f.id for f in group],
                "narrative": f"Multiple attacks detected on {host}: {len(group)} findings",
                "severity": max(f.severity for f in group).value,
            })
    return correlations
```
