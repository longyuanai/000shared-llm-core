```python
"""轻量符号执行(基于约束求解,不依赖 z3)。

简单 if/loop/算术约束求解。
"""
from __future__ import annotations
from dataclasses import dataclass


@dataclass(frozen=True)
class SymbolicConstraint:
    """形如 'x > 5' 的简单约束。"""
    var: str
    op: str  # > < >= <= == !=
    value: int | float


@dataclass(frozen=True)
class SymbolicPath:
    constraints: tuple[SymbolicConstraint, ...]
    feasible: bool


def solve_path(constraints: list[SymbolicConstraint]) -> SymbolicPath:
    """简化版:检查约束间是否矛盾(同 var 不同 op/value)。
    
    若矛盾 → feasible=False
    """
    by_var: dict[str, list[SymbolicConstraint]] = {}
    for c in constraints:
        by_var.setdefault(c.var, []).append(c)
    
    feasible = True
    for var, cs in by_var.items():
        # 简化:检查是否有 x > 5 和 x < 3 同时存在
        upper = [c for c in cs if c.op in ("<", "<=")]
        lower = [c in cs for c in cs if c.op in (">", ">=")]
        if upper and lower:
            for u in upper:
                for l in lower:
                    if l.value > u.value:
                        feasible = False
                        break
    return SymbolicPath(constraints=tuple(constraints), feasible=feasible)
```

```python
"""基线 vs 当前二进制对比,识别函数级别 + 指令级别 diff。"""
from __future__ import annotations
from dataclasses import dataclass


@dataclass(frozen=True)
class FunctionDiff:
    name: str
    baseline_size: int
    current_size: int
    instructions_added: tuple[str, ...]
    instructions_removed: tuple[str, ...]


def diff_binaries(
    baseline_funcs: dict[str, list[str]],
    current_funcs: dict[str, list[str]],
) -> list[FunctionDiff]:
    """简化:基线 vs 当前 函数表对比。
    
    baseline_funcs / current_funcs: {func_name: [instr_lines]}
    """
    all_funcs = set(baseline_funcs) | set(current_funcs)
    diffs: list[FunctionDiff] = []
    for name in sorted(all_funcs):
        base = baseline_funcs.get(name, [])
        curr = current_funcs.get(name, [])
        added = tuple(sorted(set(curr) - set(base)))
        removed = tuple(sorted(set(base) - set(curr)))
        if added or removed:
            diffs.append(FunctionDiff(
                name=name,
                baseline_size=len(base),
                current_size=len(curr),
                instructions_added=added,
                instructions_removed=removed,
            ))
    return diffs
```

```python
"""加密常量识别(基于已知 magic bytes)。"""
from __future__ import annotations
from dataclasses import dataclass


# 常见加密算法 magic bytes / 常量片段
KNOWN_CRYPTO_CONSTANTS = {
    "AES_SBOX": bytes.fromhex("637c777bf26b6fc53001672bfed7ab76"),
    "SHA256_K": bytes.fromhex("428a2f98d728ae2271374491230dbf4c"),
    "MD5_INIT": bytes.fromhex("67452301efcdab89"),
    "DES_SBOX": bytes.fromhex("0e040d020b09080a"),
}


@dataclass(frozen=True)
class CryptoMatch:
    algorithm: str
    offset: int
    length: int


def identify_crypto(binary: bytes) -> list[CryptoMatch]:
    """扫描 binary,找加密算法常量。"""
    matches: list[CryptoMatch] = []
    for algo, const in KNOWN_CRYPTO_CONSTANTS.items():
        idx = 0
        while True:
            pos = binary.find(const, idx)
            if pos < 0:
                break
            matches.append(CryptoMatch(algorithm=algo, offset=pos, length=len(const)))
            idx = pos + 1
    return matches
```
