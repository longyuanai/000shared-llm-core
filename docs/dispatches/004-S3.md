```python
"""基于 tree-sitter 的 taint analysis。

Source: 用户输入函数(input/argv/HTTP params)
Sink: 危险函数(eval/exec/SQL/RCE)
"""
from __future__ import annotations
from dataclasses import dataclass
from shared_llm_core import (
    Finding, FindingSource, FindingSeverity,
    Rule, RuleContext,
)


@dataclass(frozen=True)
class TaintPath:
    source: str
    sink: str
    path: tuple[str, ...]


class TaintAnalysisRule(Rule):
    id = "004-taint-flow"
    severity_hint = "high"

    def evaluate(self, ctx: RuleContext) -> list[Finding]:
        paths = ctx.facts.get("taint_paths", [])
        if not paths:
            return []
        results = []
        for p in paths:
            if not isinstance(p, TaintPath):
                continue
            results.append(Finding(
                id="",
                source=FindingSource.CODE,
                severity=FindingSeverity.HIGH,
                confidence=0.8,
                title=f"Taint flow: {p.source} → {p.sink}",
                host=ctx.subject,
                evidence=(f"path={' → '.join(p.path)}",),
                tags=frozenset(["taint", "data-flow"]),
            ))
        return results
```

### ISSUE 2 · DATAFLOW-001-A 跨函数数据流

```python
"""跨函数数据流跟踪。

函数 A 返回值作为函数 B 的参数 → 自动跟踪。
"""
from __future__ import annotations
from dataclasses import dataclass


@dataclass(frozen=True)
class CrossFunctionCall:
    caller: str
    callee: str
    arg_index: int


@dataclass(frozen=True)
class DataFlowChain:
    """跨函数调用链 + taint 标记。"""
    calls: tuple[CrossFunctionCall, ...]
    tainted: bool


def trace_dataflow(
    call_graph: dict[str, list[str]],
    entry: str,
    *,
    max_depth: int = 10,
) -> DataFlowChain:
    """简化 BFS:从 entry 出发,trace 调用链。"""
    visited: set[str] = set()
    queue: list[tuple[str, int]] = [(entry, 0)]
    chain: list[CrossFunctionCall] = []
    while queue and len(chain) < max_depth:
        current, depth = queue.pop(0)
        if current in visited or depth > max_depth:
            continue
        visited.add(current)
        for callee in call_graph.get(current, []):
            chain.append(CrossFunctionCall(caller=current, callee=callee, arg_index=0))
            queue.append((callee, depth + 1))
    return DataFlowChain(calls=tuple(chain), tainted=any("user_input" in c.callee for c in chain))
```

```python
"""把 dataflow 路径转成自然语言 prompt,调 v0.1 LLMRouter 解释。"""
from __future__ import annotations
from shared_llm_core import ChatRequest, ChatMessage, LLMRouter, TaskTier
from shared_llm_core.finding import Finding, FindingSource, FindingSeverity

from src.codeguard.dataflow import DataFlowChain


def explain_chain(
    router: LLMRouter,
    chain: DataFlowChain,
    file_path: str,
) -> Finding:
    """调 cheap tier LLM,生成 human-readable 解释。"""
    path_str = " → ".join(f"{c.caller}()→{c.callee}()" for c in chain.calls)
    prompt = (
        f"Explain this dataflow path in 1-2 sentences:\n"
        f"File: {file_path}\n"
        f"Path: {path_str}\n"
        f"Tainted: {chain.tainted}"
    )
    req = ChatRequest(
        model="stub",
        messages=[ChatMessage(role="user", content=prompt)],
        response_format={"type": "json_object"},
    )
    resp = router.chat(TaskTier.CHEAP, req)
    explanation = resp.choices[0].message.content
    return Finding(
        id="",
        source=FindingSource.CODE,
        severity=FindingSeverity.MEDIUM if chain.tainted else FindingSeverity.LOW,
        confidence=0.7,
        title=f"Dataflow explanation for {file_path}",
        host=file_path,
        evidence=(explanation,),
        tags=frozenset(["dataflow", "llm-explain"]),
    )
```
