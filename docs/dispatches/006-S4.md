### ISSUE 2 · ZERODAY-001-A 0-day 预测

**新建** `src/ai_firmware_agent/zeroday.py`:

```python
"""基于组件版本 + 已知 CVE 模式做 0-day 预测。"""
from __future__ import annotations
from dataclasses import dataclass


@dataclass(frozen=True)
class ZeroDayCandidate:
    component: str
    version: str
    similar_cve: str
    confidence: float
    reason: str


# Demo:已知 CVE 模式库
CVE_PATTERN_LIBRARY = {
    "CVE-2021-36260": {"component": "hikvision", "pattern": "web server command injection"},
    "CVE-2017-7921": {"component": "hikvision", "pattern": "authentication bypass"},
    "CVE-2018-10561": {"component": "d-link", "pattern": "command injection"},
}


def predict_zeroday(component: str, version: str) -> list[ZeroDayCandidate]:
    """基于 component 名匹配 CVE 模式库,生成候选 0-day。"""
    candidates: list[ZeroDayCandidate] = []
    for cve, info in CVE_PATTERN_LIBRARY.items():
        if info["component"] == component.lower():
            candidates.append(ZeroDayCandidate(
                component=component,
                version=version,
                similar_cve=cve,
                confidence=0.6,
                reason=f"Similar to {cve} ({info['pattern']})",
            ))
    return candidates
```

"""从 firmware 漏洞到 root shell 攻击链还原。

用 §7 MultiAgentOrchestrator:
- SCOUT: 找漏洞组件
- EXPLOITER: 模拟 exploit
- REVIEWER: 评估影响
"""
from __future__ import annotations
from shared_llm_core import (
    AgentRole, AgentResult, MissionContext, MultiAgentOrchestrator,
)


def trace_attack_chain(
    orch: MultiAgentOrchestrator,
    firmware_components: list[dict],
) -> list[AgentResult]:
    mission = MissionContext(
        task="Trace attack chain from firmware to root shell",
        inputs={"components": firmware_components},
    )
    return orch.run(
        mission,
        [AgentRole.SCOUT, AgentRole.EXPLOITER, AgentRole.REVIEWER],
    )
