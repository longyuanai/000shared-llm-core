"""基于资产拓扑调整 CVSS:
- public-facing 资产: CVSS +0.5
- internal-only 资产: CVSS -0.3
"""
from shared_llm_core import (
    Finding, FindingSource, FindingSeverity,
    Rule, RuleContext,
)


class TopologyAwareCVSSRule(Rule):
    id = "002-topology-cvss"
    severity_hint = "high"

    def evaluate(self, ctx: RuleContext) -> list[Finding]:
        topology = ctx.facts.get("topology", "internal")
        base_cvss = ctx.facts.get("base_cvss", 0.0)
        if not isinstance(base_cvss, (int, float)):
            return []

        if topology == "public":
            adjusted = min(10.0, base_cvss + 0.5)
        elif topology == "internal":
            adjusted = max(0.0, base_cvss - 0.3)
        else:
            adjusted = base_cvss

        return [Finding(
            id="",
            source=FindingSource.VULN,
            severity=FindingSeverity.HIGH if adjusted >= 7.0 else FindingSeverity.MEDIUM,
            confidence=0.85,
            title=f"Topology-adjusted CVSS for {ctx.subject}",
            host=ctx.subject,
            evidence=(f"topology={topology}", f"base_cvss={base_cvss}", f"adjusted={adjusted}"),
            tags=frozenset(["topology", "cvss-adjust"]),
        )]
```

"""ExploitDB / Metasploit 模块库接入(离线 JSON + httpx mock)。"""
from __future__ import annotations
from pathlib import Path
import json
import httpx


class ExploitDB:
    """ExploitDB 离线索引。
    
    实际生产可换 https://www.exploit-db.com,但 v0.5 用离线 JSON 即可。
    """

    def __init__(self, index_path: Path | str | None = None) -> None:
        self._index: dict[str, dict] = {}
        if index_path:
            self.load(Path(index_path))

    def load(self, path: Path) -> None:
        self._index = json.loads(path.read_text(encoding="utf-8"))

    def lookup(self, cve: str) -> dict | None:
        return self._index.get(cve)

    def has_exploit(self, cve: str) -> bool:
        return cve in self._index


class MetasploitIndex:
    """Metasploit 模块名索引(简化版)。"""

    def __init__(self) -> None:
        self._modules: dict[str, list[str]] = {}

    def register(self, cve: str, module_name: str) -> None:
        self._modules.setdefault(cve, []).append(module_name)

    def modules_for(self, cve: str) -> list[str]:
        return list(self._modules.get(cve, []))


def enrich_with_exploit(
    finding_dict: dict,
    exploit_db: ExploitDB,
    metasploit: MetasploitIndex,
) -> dict:
    """给 finding 加 exploit_available + module_name 字段。"""
    cve = finding_dict.get("cve", "")
    if not cve:
        return finding_dict
    enriched = dict(finding_dict)
    enriched["exploit_available"] = exploit_db.has_exploit(cve)
    enriched["metasploit_modules"] = metasploit.modules_for(cve)
    return enriched
```

### ISSUE 3 · TIMELINE-002-A CVE 时间线

**新建** `src/ai_vuln_agent/timeline.py`:

```python
"""基于 EPSS 历史生成 CVE 披露时间线。"""
from __future__ import annotations
from dataclasses import dataclass
from datetime import datetime


@dataclass(frozen=True)
class CVETimeline:
    cve: str
    first_seen: datetime
    peak_epss: float
    days_to_peak: int


def compute_timeline(cve: str, epss_history: list[tuple[datetime, float]]) -> CVETimeline | None:
    """epss_history 是按时间排序的 (timestamp, epss_score) 列表。

    返回 first_seen / peak / days_to_peak。
    """
    if not epss_history:
        return None
    first_seen = epss_history[0][0]
    peak_score, peak_ts = max(epss_history, key=lambda x: x[1])
    days_to_peak = (peak_ts - first_seen).days
    return CVETimeline(
        cve=cve,
        first_seen=first_seen,
        peak_epss=peak_score,
        days_to_peak=days_to_peak,
    )
```
